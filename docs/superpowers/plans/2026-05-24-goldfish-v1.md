# Goldfish v1.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Close the gap between passive memory (hooks/drain/vault — already working) and active memory (MCP servers for GitNexus, OMEGA, Semble), stabilize the hook binary path, update the CLAUDE.md active layer block to reference MCP tool names, improve wake-up context quality, and complete operational tooling (doctor, replay, semble_indexed_at tracking).

**Architecture:** `init.py` gains a self-install step (ensuring `~/.local/bin/goldfishh` is the stable hook target) and a MCP registration block (calling each tool's own setup CLI). `drain.py` gains vault reads for open tasks and recent decisions in wake-up generation, plus `semble_indexed_at` manifest writes after indexing. `cli.py`'s `doctor` gains three new checks, and `replay` becomes resumable from `last_byte_offset`.

**Tech Stack:** Python 3.11+, pytest, unittest.mock, typer, pathlib, subprocess, PyYAML, tomli-w

**Spec:** `docs/superpowers/specs/2026-05-24-goldfishh-v1-design.md`

def _detect_goldfish_bin() -> str:
    stable = Path.home() / ".local" / "bin" / "goldfishh"
    if stable.exists():

    found = shutil.which("goldfishh")
    if found:

```python
@app.command()
def replay() -> None:
    """Rebuild vault from Claude Code JSONL transcripts. Resumable."""
    cwd = os.getcwd()
    project = project_name(cwd)
    encoded = cwd.replace("/", "-")
    jsonl_dir = Path.home() / ".claude" / "projects" / encoded

    if not jsonl_dir.exists():
        typer.echo(f"No transcript directory found at {jsonl_dir}")
        raise typer.Exit(1)

    manifest = get_manifest(project)
    resume_file = manifest.get("last_jsonl_file", "")
    resume_offset = int(manifest.get("last_byte_offset", 0))
    total = 0

    # Sort by mtime oldest-first for chronological replay
    jsonl_files = sorted(jsonl_dir.glob("*.jsonl"), key=lambda f: f.stat().st_mtime)

    # Determine the mtime cutoff: skip files strictly older than resume_file
    resume_mtime = 0.0
    if resume_file:
        resume_path = jsonl_dir / resume_file
        if resume_path.exists():
            resume_mtime = resume_path.stat().st_mtime

    for jsonl_file in jsonl_files:
        file_mtime = jsonl_file.stat().st_mtime
        if resume_mtime and file_mtime < resume_mtime:
            typer.echo(f"Skipping {jsonl_file.name} (already processed)")
            continue

        start_offset = resume_offset if jsonl_file.name == resume_file else 0
        typer.echo(f"Replaying {jsonl_file.name} from byte {start_offset}...")

        with jsonl_file.open("rb") as f:
            f.seek(start_offset)
            for raw_line in f:
                line = raw_line.decode(errors="replace").strip()
                if not line:
                    continue
                try:
                    event = json.loads(line)
                    drain._route(event)
                    total += 1
                except Exception:
                    pass
            offset = f.tell()

        write_manifest(
            project,
            {**manifest, "last_byte_offset": offset, "last_jsonl_file": jsonl_file.name,
             "bootstrap_complete": True},
        )
        manifest = get_manifest(project)
        resume_offset = 0  # Only seek on the resume file; subsequent files read from byte 0

    typer.echo(f"Replay complete. Processed {total} events.")
```
